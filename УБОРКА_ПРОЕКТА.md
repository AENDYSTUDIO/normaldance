## 🚀 АВТОМАТИЧЕСКАЯ ОЧИСТКА

### 📋 **Шаг 1: Создание скрипта очистки**
```bash
# Создать скрипт очистки
cat > scripts/cleanup-project.sh << 'EOF'
#!/bin/bash
# NORMAL DANCE Project Cleanup Script
# Удаляет документацию с результатами и планами

set -e

RED='\033[0;31m'
GREEN='\033[0;32m'
YELLOW='\033[1;33m'
NC='\033[0m'

log_info() {
    echo -e "${GREEN}[INFO]${NC} $1"
}

log_warn() {
    echo -e "${YELLOW}[WARN]${NC} $1"
}

log_error() {
    echo -e "${RED}[ERROR]${NC} $1"
}

echo -e "${GREEN}🧹 NORMAL DANCE Project Cleanup${NC}"
echo -e "${GREEN}================================${NC}"
echo

log_info "Starting project cleanup..."
log_warn "This will remove plans, reports, and marketing materials."

# Удалить все файлы отчетов и планов
log_info "Removing reports and plans..."
rm -f FINAL_REPORT.md PFINAL_SUMMARY.md RELEASE_SUMMARY.md
rm -f RELEASE-NOTES.md LAUNCH_REPORT.md DAY1_SUMMARY.md
rm -f IMPLEMENTATION_PLAN.md PROJECT_ANALYSIS_REPORT.md
rm -f PROJECT_IMPROVEMENT_SUMMARY.md
rm -f *_summary.md *_report.md *_plan.md

# Удалить task файлы
log_info "Removing task files..."
rm -f TASK*_*.md
rm -f PHASE*_*.md
rm -f SOLO_DEV_TODO.md

# Удалить технические отчеты
log_info "Removing technical reports..."
rm -f TECHNICAL_DEBT_*.md ARCHITECTURE_ANALYSIS_REPORT.md
rm -f IMPROVEMENTS_SUMMARY.md

# Удалить маркетинговые материалы
log_warn "Removing marketing materials..."
rm -rf sales-packet/ grants/
rm -f pitch_deck.md investor-relations.md
rm -rf telegram-partnership/ risk-management/

# Удалить roadmaps и планы развертывания
log_warn "Removing roadmaps and deployment guides..."
rm -f VERCEL_DEPLOYMENT_ROADMAP.md INVISIBLE_WALLET_*.md
rm -f QUICK_START.md START_HERE.md READY_TO_RUN.md

# Удалить временные файлы
log_info "Removing temporary files..."
rm -f CREATE_REPOSITORIES.md ЗАПУСК_ПРЯМО_СЕЙЧАС.md
rm -f БЫСТРАЯ_РАЗВЕРТКА.md РАЗВЕРТЫВАНИЕ_*.md
rm -f VERCEL_DEPLOYMENT_GUIDE_RU.md brif.md блокчеин*.md

# Удалить архив документации
log_warn "Removing documentation archive..."
rm -rf _archive-docs/

# Удалить лог файлы и временные файлы
log_info "Cleaning up logs and temp files..."
rm -f *.log *.backup *generated*
rm -rf temp/ logs/

# Удалить все .md файлы в корне (оставить только README.md)
log_warn "Cleaning root markdown files..."
ls *.md | grep -v "README.md" | xargs rm -f 2>/dev/null || true

log_info "✅ Cleanup completed!"
echo
echo -e "${GREEN}📁 Project structure after cleanup:${NC}"
echo
echo -e "${BLUE}✅ Kept:${NC}"
echo "src/           - Source code"
echo "contracts/     - Smart contracts"
echo "package.json   - Dependencies"
echo
echo -e "${YELLOW}⚠️ Check that no important technical docs were removed${NC}"

if command -v tree &> /dev/null; then
    echo -e "${BLUE}📊 Current project structure:${NC}"
    tree -I 'node_modules|.next|coverage|*.log|*.backup' -L 2
fi
EOF

chmod +x scripts/cleanup-project.sh
```

### 🔄 **Шаг 2: Запустить очистку**
```bash
# Запустить скрипт очистки
./scripts/cleanup-project.sh
```
